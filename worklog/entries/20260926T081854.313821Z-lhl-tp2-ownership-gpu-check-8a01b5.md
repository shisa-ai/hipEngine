```python
import json
from hipengine.core.hip import get_hip_runtime
from hipengine.core.device import scoped_current_device
from hipengine.distributed.tp2_generate import MlpTP2GenerationSession
rt = get_hip_runtime()
allocated, freed = {}, set()
original_malloc, original_free = rt.malloc, rt.free

def allocate(n):
    p = original_malloc(n)
    allocated[int(p)] = (rt.get_device(), int(n))
    freed.discard(int(p))
    return p

def release(p):
    result = original_free(p)
    freed.add(int(p))
    return result

rt.malloc, rt.free = allocate, release

def mem():
    values = {}
    for d in (0, 1):
        with scoped_current_device(rt, d):
            rt.device_synchronize()
            values[d] = rt.mem_get_info()[0]
    return values

print('before', json.dumps(mem()), flush=True)
s = MlpTP2GenerationSession('/models/gguf/Qwen3.8-27B-Q4_K_M.gguf',
    devices=(0, 1), max_sequence_length=64)
print('loaded', json.dumps(mem()), flush=True)
s.close()
s.close()
remaining = {str(p): v for p, v in allocated.items() if p not in freed}
print(json.dumps({'after': mem(), 'unique_addresses': len(allocated),
    'unfreed_count': len(remaining),
    'unfreed_bytes': sum(v[1] for v in remaining.values()),
    'unfreed': remaining}), flush=True)
assert not remaining, remaining
```
