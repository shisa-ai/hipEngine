## Validation

- CPU check from the repository root, on runtime code `10d814df5`. The fixture uses a smaller visit plan than the real checkpoint, so derive the indices from its plan. Truncating the plan yields the raw exit from the preceding visit; running the full plan with the existing callback yields the entry seen by the diagnostic. Both paths use the same inputs and weights.

  ```bash
  OPENBLAS_NUM_THREADS=1 .venv/bin/python - <<'PY'
  from contextlib import closing
  from dataclasses import replace
  import numpy as np
  from hipengine.runtime.inkling2_mini_reference import Inkling2TextReference, _rms
  from tests.test_unit_inkling2_reference import TinyWeights, _config

  weights, cfg, ids = TinyWeights(), _config(), [0, 1]
  ref = Inkling2TextReference(cfg, weights)
  h = _rms(weights.dense('embed.weight', ids), ref.config['rms_norm_eps']) * weights.dense('embed_norm.weight')
  def exit_at(n):
      plan = replace(ref.plan, visits=ref.plan.visits[:n])
      with closing(Inkling2TextReference(cfg, weights, plan=plan).session(4)) as session:
          try:
              next(session._traverse(h.copy(), None))
          except StopIteration as done:
              raw, blocks, _ = done.value
          else:
              raise AssertionError('unexpected staged traversal')
          return raw, session._output_hidden(raw, blocks)

  with closing(ref.session(4)) as session:
      seen = {}
      session.prefill_block(ids, capture=lambda i, value: seen.setdefault(i, value.copy()))
  visits = ref.plan.visits
  entry = next(i for i, v in enumerate(visits) if v.weight_layer == ref.config['loop_config']['core_start'])
  restart = next(i for i, v in enumerate(visits) if v.restart_from_prelude)
  assert np.array_equal(seen[restart], seen[entry])
  assert not np.allclose(seen[restart], exit_at(restart)[0])
  raw, final = exit_at(len(visits))
  assert np.array_equal(seen[len(visits)], final)
  assert not np.allclose(seen[len(visits)], raw)
  print('Restart captures the restored entry; terminal capture includes final depth mix and normalization.')
  PY
  ```
